    <script>
        // State management
        let currentStep = 1;

        // --- DOM Elements ---
        const motifOptions = document.querySelectorAll('.motif-option');
        const motifInput = document.getElementById('motif_selector');
        const motifTextarea = document.getElementById('motif_intention');

        // --- Step 1 Logic: Motif Selection ---
        motifOptions.forEach(option => {
            option.addEventListener('click', function() {
                // Remove selected class from all
                motifOptions.forEach(opt => opt.classList.remove('selected'));
                // Add to clicked
                this.classList.add('selected');
                // Set input value
                motifInput.value = this.getAttribute('data-value');

                // Optional: If user wants text in textarea to start with selection
                // if(motifTextarea.value === '') motifTextarea.value = this.getAttribute('data-value') + ": ";
            });
        });

        // --- Navigation Logic ---
        function showStep(step) {
            // Hide all steps
            document.querySelectorAll('.wizard-step').forEach(el => el.classList.remove('active'));
            // Show target step
            document.getElementById(`step-${step}`).classList.add('active');

            // Update Stepper UI
            for (let i = 1; i <= 3; i++) {
                const ind = document.getElementById(`step-indicator-${i}`);
                ind.classList.remove('active', 'completed');
                if (i === step) ind.classList.add('active');
                if (i < step) ind.classList.add('completed');
            }

            currentStep = step;
            window.scrollTo(0, 0);
        }

        function nextStep(step) {
            if (step === 2) {
                // Validate Step 1
                if (!motifInput.value && !motifTextarea.value) {
                    alert("Veuillez sélectionner un motif ou entrer des précisions.");
                    return;
                }
                // Merge motif for backend if needed. 
                // Backend expects 'motif_intention'. We have 'motif_selector' (new) and 'motif_intention' (details).
                // We should ensure 'motif_intention' contains everything if backend only checks that one field.
                // But let's leave them separate in DOM and assume User fills details.
                // If 'motif_intention' is required by backend validation, we must ensure it's not empty.
                if (motifTextarea.value.trim() === "") {
                    if (motifInput.value) {
                        // If user only selected a button, copy it to textarea to satisfy backend 'required' check if it exists
                        // Or rely on backend validation failing and showing error.
                        // Let's AUTO-FILL textarea if empty
                        motifTextarea.value = motifInput.value;
                    } else {
                        alert("Veuillez préciser votre intention.");
                        return;
                    }
                }
            }

            showStep(step);
            if (step === 3) generateRecap();
        }

        function prevStep(step) {
            showStep(step);
        }

        function validateStep2() {
            const p = document.getElementById('paroisse_id').value;
            const c = document.getElementById('celebration_choisie').value;
            const mode = document.getElementById('type_date_selection').value;

            if (!p || !c) {
                alert("Veuillez choisir une paroisse et un type de célébration.");
                return;
            }

            if (mode === 'date') {
                const d = document.getElementById('date_souhaitee').value;
                const h = document.getElementById('heure_souhaitee').value;
                if (!d || !h) {
                    alert("Veuillez sélectionner une date et une heure.");
                    return;
                }
            } else {
                const m = document.getElementById('mois_souhaite').value;
                if (!m) {
                    alert("Veuillez sélectionner un mois.");
                    return;
                }
            }

            nextStep(3);
        }

        // --- Recap Logic ---
        function generateRecap() {
            const container = document.getElementById('recap-container');

            // Gather Data
            const motif = document.getElementById('motif_selector').value || "Autre";
            const details = document.getElementById('motif_intention').value;
            const intercession = document.getElementById('interception_par').value || "Aucune";

            const villeSel = document.getElementById('ville_id');
            const villeText = villeSel.options[villeSel.selectedIndex]?.text;

            const paroisseSel = document.getElementById('paroisse_id');
            const paroisseText = paroisseSel.options[paroisseSel.selectedIndex]?.text;

            const mode = document.getElementById('type_date_selection').value;
            const type = document.getElementById('celebration_choisie').value;
            const montant = document.getElementById('montant_offrande').value;

            let dateDisplay = "";
            if (mode === 'date') {
                const date = document.getElementById('date_souhaitee').value;
                const heure = document.getElementById('heure_souhaitee').value;
                dateDisplay = `<div class="recap-row"><span>Date</span><span>${date} à ${heure}</span></div>`;
            } else {
                const mois = document.getElementById('mois_souhaite').value;
                dateDisplay = `<div class="recap-row"><span>Mois</span><span>${mois}</span></div>`;
            }

            container.innerHTML = `
            <div class="recap-box">
                <h3>Détails de votre demande</h3>
                <div class="recap-row"><span>Motif</span><span>${motif}</span></div>
                <div class="recap-row"><span>Précision</span><span>${details}</span></div>
                <div class="recap-row"><span>Intercession</span><span>${intercession}</span></div>
            </div>
            
            <div class="recap-box">
                <h3>Détails de la Messe</h3>
                <div class="recap-row"><span>Paroisse</span><span>${paroisseText} (${villeText})</span></div>
                ${dateDisplay}
                <div class="recap-row"><span>Type</span><span>${type}</span></div>
                <div class="recap-row"><span>Montant offrande</span><span>${montant} FCFA</span></div>
            </div>
        `;
        } // End Recap Logic

        // --- Toggle Date/Month Logic ---
        const dateOptions = document.querySelectorAll('.type-date-option');
        const typeLinkInput = document.getElementById('type_date_selection');
        const dateContainer = document.getElementById('date-specific-container');
        const monthContainer = document.getElementById('month-specific-container');

        dateOptions.forEach(opt => {
            opt.addEventListener('click', function() {
                dateOptions.forEach(o => o.classList.remove('selected'));
                this.classList.add('selected');

                typeLinkInput.value = this.getAttribute('data-value');
                if (typeLinkInput.value === 'date') {
                    dateContainer.style.display = 'block';
                    monthContainer.style.display = 'none';

                    // Add required attribute dynamically for browser validation if needed, 
                    // though we use custom validation in validateStep2
                } else {
                    dateContainer.style.display = 'none';
                    monthContainer.style.display = 'block';
                }
            });
        });


        // --- Dynamic Select Loading (Legacy + New) ---
        // Copied and adapted from previous file logic
        const villeSelect = document.getElementById('ville_id');
        const communeSelect = document.getElementById('commune_id');
        const paroisseSelect = document.getElementById('paroisse_id');
        const montantInput = document.getElementById('montant_offrande');
        const montantDetails = document.getElementById('montant-details');

        let montantUnitaire = 0;

        villeSelect.addEventListener('change', function() {
            const villeId = this.value;
            resetSelect(communeSelect, "Sélectionnez d'abord une ville");
            resetSelect(paroisseSelect, "Sélectionnez d'abord une commune");

            if (villeId) {
                fetch(`/get-communes/${villeId}`)
                    .then(r => r.json())
                    .then(data => populateSelect(communeSelect, data, 'Sélectionnez une commune', 'id',
                        'nom_commune'));
            }
        });

        communeSelect.addEventListener('change', function() {
            const comId = this.value;
            resetSelect(paroisseSelect, "Sélectionnez d'abord une commune");
            if (comId) {
                fetch(`/get-paroisses/${comId}`)
                    .then(r => r.json())
                    .then(data => populateSelect(paroisseSelect, data, 'Sélectionnez une paroisse', 'id', 'name', {
                        'data-montant': 'montant_offrande'
                    }));
            }
        });

        paroisseSelect.addEventListener('change', function() {
            const opt = this.options[this.selectedIndex];
            montantUnitaire = parseFloat(opt.getAttribute('data-montant')) || 0;
            updateTotal();
        });

        // Existing helper functions
        function resetSelect(el, text) {
            el.innerHTML = `<option value="">${text}</option>`;
            el.disabled = true;
        }

        function populateSelect(el, data, defaultText, valKey, textKey, attrs = {}) {
            let opts = `<option value="">${defaultText}</option>`;
            data.forEach(item => {
                let attrStr = '';
                for (let k in attrs) attrStr += ` ${k}="${item[attrs[k]]}"`;
                opts += `<option value="${item[valKey]}"${attrStr}>${item[textKey]}</option>`;
            });
            el.innerHTML = opts;
            el.disabled = false;
        }

        // Simplified calculation for now (assuming base price * 1 for simplicity until day selection is re-added if valid)
        // NOTE: The previous file had day selection. The screenshot DOES NOT show checkbox for every day. 
        // It implies a single date selection "Date".
        // IF user wants multiple days, they might need the check boxes again. 
        // BUT the prompt asked to follow the screenshot which shows "Date" and "Heure" inputs (Singular).
        // I will stick to singular date for now.
        function updateTotal() {
            montantInput.value = montantUnitaire;
            if (montantUnitaire > 0) montantDetails.textContent = `Tarif unitaire : ${montantUnitaire} FCFA`;
        }

        // Set min date
        document.getElementById('date_souhaitee').min = new Date().toISOString().split('T')[0];
    </script>
@endsection
